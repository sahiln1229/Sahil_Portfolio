import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split

class CropRecommendationPipeline:
    def __init__(self, data_path: str):
        self.df = pd.read_csv(data_path)
        self.features = ['N', 'P', 'K', 'temperature', 'humidity', 'ph', 'rainfall']
        self.model = RandomForestClassifier(n_estimators=100, random_state=42)

    def train_and_evaluate(self):
        X = self.df[self.features]
        y = self.df['label']
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)
        
        self.model.fit(X_train, y_train)
        accuracy = self.model.score(X_test, y_test)
        print(f"[ML_ACCURACY] Test Score: {accuracy * 100:.2f}%")
        return accuracy

#include <Arduino.h>

const int MQ3_PIN = A0;
const int RELAY_PIN = 7;
const int BUZZER_PIN = 8;
const int THRESHOLD = 350;

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // Armed
}

void loop() {
  int sensorValue = analogRead(MQ3_PIN);
  if (sensorValue > THRESHOLD) {
    digitalWrite(RELAY_PIN, LOW); // Interlock Cutoff
    digitalWrite(BUZZER_PIN, HIGH);
    Serial.println("[ALERT] Alcohol detected. Ignition locked.");
  } else {
    digitalWrite(RELAY_PIN, HIGH);
    digitalWrite(BUZZER_PIN, LOW);
  }
  delay(100);
}
